# 队费记账（比赛结算扣款 + 站内通知）实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 队员交队费 → 赛后按出场名单结算扣款（记流水、允许欠款为负）→ 余额 ≤0 站内通知，前后端完整闭环。

**Architecture:** Go 新建 `internal/teamfund`（结算+队费流水）与 `internal/notification`（站内通知）两个六边形模块；match 模块新增只读的结算名单服务；payment 的队费核销追加流水。小程序重写结算表单（固定名单+每人金额）、改造记账页、修正旧支付路径。设计详见 `docs/superpowers/specs/2026-08-23-team-fund-settlement-design.md`。

**Tech Stack:** Go + Gin + PostgreSQL + pgx + sqlc + goose；uni-app + Vue 3 + TS。

## Global Constraints

- 兼容性硬规则：只新增路由/表/字段，不删不改既有路由与响应字段；已发布小程序既有调用不得由成功变失败。
- 后端分层：handler 只做协议适配/DTO；业务规则在 domain/application；SQL 在 adapters/postgres + db/queries。
- Gin 路由：比赛相关新路由参数名必须用 `:id`（与 match 模块既有 `/matches/:id` 一致，否则 gin 启动 panic）。
- 金额一律整数分（`*_cents`，BIGINT）；team_fund_transactions.amount_cents 带符号（正=入账，负=扣费）。
- Go module 路径前缀：`github.com/oryjk/registration_system/registration_system_go`（下文简写 `<go>`）。
- 提交前验证：`cd <go> && gofmt -w . && go vet ./... && go test -race ./... && go build -o /tmp/registration-system-go-api ./cmd/api`；小程序 `bun run type-check`（必要时 `bun run build:mp-weixin`）。
- 工作区有用户未提交改动：**本计划各任务不做 git commit**，除非用户明确要求。
- sqlc 生成命令：`sqlc generate`（在 `<go>` 目录）。

```go
	Title       string
	Content     string

	RelatedID   string

	CreatedAt   time.Time

func NewNotification(userID int64, kind, title, content, relatedType, relatedID string, now time.Time) (Notification, error) {
	kind = strings.TrimSpace(kind)
	title = strings.TrimSpace(title)
	content = strings.TrimSpace(content)
	if userID <= 0 || kind == "" || title == "" || content == "" {
		return Notification{}, sharederror.New(sharederror.KindValidation, "通知参数无效")
	}
	return Notification{UserID: userID, Kind: kind, Title: title, Content: content,
		RelatedType: strings.TrimSpace(relatedType), RelatedID: strings.TrimSpace(relatedID), CreatedAt: now}, nil
}
```

```go
package application

import (
	"context"

	sharedauth "<go>/internal/shared/auth"
	sharederror "<go>/internal/shared/domain"
	notificationdomain "<go>/internal/notification/domain"
	notificationports "<go>/internal/notification/ports"
)

// SystemNotification 系统事件通知入参（无用户会话，由各业务模块调用）。
type SystemNotification struct {
	UserID      int64
	Kind        string
	Title       string
	Content     string
	RelatedType string
	RelatedID   string
}

type ListQuery struct {
	UnreadOnly bool
	Limit      int
}

type Service struct {
	repository notificationports.Repository
	now        func() time.Time
}

func NewService(repository notificationports.Repository) *Service {
	return &Service{repository: repository, now: time.Now}
}

func (s *Service) Notify(ctx context.Context, message SystemNotification) error {
	notification, err := notificationdomain.NewNotification(message.UserID, message.Kind, message.Title,
		message.Content, message.RelatedType, message.RelatedID, s.now())
	if err != nil {
		return err
	}
	_, err = s.repository.Create(ctx, notification)
	return err
}

func (s *Service) List(ctx context.Context, actor sharedauth.Actor, query ListQuery) ([]notificationdomain.Notification, error) {
	if !actor.IsUser() {
		return nil, sharederror.ErrForbidden
	}
	limit := query.Limit
	if limit <= 0 || limit > 100 {
		limit = 50
	}
	return s.repository.List(ctx, notificationports.ListFilter{UserID: actor.ID, UnreadOnly: query.UnreadOnly, Limit: limit})
}

func (s *Service) UnreadCount(ctx context.Context, actor sharedauth.Actor) (int64, error) {
	if !actor.IsUser() {
		return 0, sharederror.ErrForbidden
	}
	return s.repository.CountUnread(ctx, actor.ID)
}

func (s *Service) MarkAllRead(ctx context.Context, actor sharedauth.Actor) (int64, error) {
	if !actor.IsUser() {
		return 0, sharederror.ErrForbidden
	}
	return s.repository.MarkAllRead(ctx, actor.ID)
}
```

```go
type SettlementRosterSource interface {

func (s *SettlementRosterService) LoadSettlementRoster(ctx context.Context, matchID uuid.UUID) (SettlementRoster, bool, error) {
	match, groups, found, err := s.repository.FindByID(ctx, matchID) // 既有方法
	if err != nil || !found { return SettlementRoster{}, false, err }
	teamByGroupID := map[uuid.UUID]int64{}
	for _, g := range groups {
		if g.TeamID != nil { teamByGroupID[g.ID] = *g.TeamID }

	rows, err := s.repository.ListSettlementAttendees(ctx, matchID)
	if err != nil { return SettlementRoster{}, false, err }
	attendees := make([]SettlementAttendee, 0, len(rows))

		attendees = append(attendees, SettlementAttendee{

			TeamID: teamByGroupID[row.GroupID], Paid: row.Paid,
		}) // GroupID 需在 SQL 里带出：SELECT 加 g.id AS group_id
	}
	return SettlementRoster{MatchID: match.ID, MatchName: match.Name, Status: match.Status,
		PaymentMode: match.PaymentMode, FeePerPersonCents: match.FeePerPersonCents,
		HostTeamID: match.HostTeamID, AwayTeamID: match.AwayTeamID, Attendees: attendees}, true, nil
```

```go
package domain

import (
	"sort"

	sharederror "<go>/internal/shared/domain"
)

type Attendee struct {
	UserID   int64
	Nickname string
	TeamID   int64
	Paid     bool
}

type Charge struct {
	TeamID      int64
	UserID      int64
	AmountCents int64
}

// BuildCharges 归一化结算名单：items 必须与「有球队且未预付」的出场者完全一致；
// 金额 >= 0（0=免付），至少一人 > 0；结果按 (TeamID, UserID) 排序以便加锁顺序稳定。
func BuildCharges(attendees []Attendee, items map[int64]int64) ([]Charge, error) {
	eligible := make([]Attendee, 0, len(attendees))
	for _, attendee := range attendees {
		if attendee.TeamID != 0 && !attendee.Paid {
			eligible = append(eligible, attendee)
		}
	}
	if len(items) != len(eligible) {
		return nil, sharederror.New(sharederror.KindValidation, "结算名单与可扣名单不一致")
	}
	charges := make([]Charge, 0, len(eligible))
	total := int64(0)
	for _, attendee := range eligible {
		amount, ok := items[attendee.UserID]
		if !ok || amount < 0 {
			return nil, sharederror.New(sharederror.KindValidation, "结算金额无效")
		}
		total += amount
		charges = append(charges, Charge{TeamID: attendee.TeamID, UserID: attendee.UserID, AmountCents: amount})
	}
	if total <= 0 {
		return nil, sharederror.New(sharederror.KindValidation, "结算总额需要大于 0")
	}
	sort.Slice(charges, func(i, j int) bool {
		if charges[i].TeamID != charges[j].TeamID { return charges[i].TeamID < charges[j].TeamID }
		return charges[i].UserID < charges[j].UserID
	})
	return charges, nil
}
```

```go
func (r *Repository) SettleInTransaction(ctx, matchID, createdByUserID, description, charges) (SettleOutcome, error) {
	tx := r.database.Begin(ctx); defer rollback
	q := r.queries.WithTx(tx)
	// 1) 串行化：锁当前生效 settle 批（无则跳过）
	active, err := q.GetActiveSettlementBatchForUpdate(ctx, matchID) // ErrNoRows → 无
	nextNo := q.GetNextSettlementBatchNo(ctx, matchID)
	reversed := int64(0)
	if active 存在 {
		reverseRow := q.InsertSettlementBatch(matchID, nextNo, "reverse", active.ID, nil,
			"冲正批次 #" + strconv.FormatInt(active.BatchNo, 10), -active.TotalAmountCents, active.UserCount, createdByUserID)
		reversed = nextNo
		nextNo++
		for _, txn := range q.ListTeamFundTransactionsBySource(ctx, "match_settlement", strconv.FormatInt(active.ID, 10)) {
			balance := q.CreditTeamMemberFund(-txn.AmountCents, txn.TeamID, txn.UserID) // 扣费为负 → 回加正数
			q.InsertTeamFundTransaction(txn.TeamID, txn.UserID, -txn.AmountCents, balance,
				"settlement_reversal", strconv.FormatInt(reverseRow.ID, 10), txn.MatchID, "结算冲正回加")
		}
		q.MarkSettlementBatchReversed(reverseRow.ID, active.ID)
	}
	// 2) 新 settle 批
	settleRow := q.InsertSettlementBatch(matchID, nextNo, "settle", nil, nil, description, total, userCount, createdByUserID)
	items := []SettlementItem{}
	for _, c := range charges { // charges 已按 (team,user) 排序（domain.BuildCharges 保证）
		if c.AmountCents == 0 { items = append(items, ...AmountCents 0, BalanceAfter 查询当前余额); continue }
		q.EnsureTeamMemberFundRow(c.TeamID, c.UserID)
		q.LockTeamMemberFund(c.TeamID, c.UserID)
		balance := q.DebitTeamMemberFund(c.AmountCents, c.TeamID, c.UserID)
		q.InsertTeamFundTransaction(c.TeamID, c.UserID, -c.AmountCents, balance,
			"match_settlement", strconv.FormatInt(settleRow.ID, 10), matchID, description)
		items = append(items, SettlementItem{c.TeamID, c.UserID, c.AmountCents, balance})
	}
	tx.Commit(ctx)
	return SettleOutcome{BatchNo: nextNo, ReversedBatchNo: reversed, ...}
}
```
